cmake_minimum_required(VERSION 3.8)
project(hnu_gazebo_adapter)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
add_library(hnu_drive_watchdog SHARED src/drive_watchdog.cpp)
target_compile_features(hnu_drive_watchdog PUBLIC cxx_std_17)
target_link_libraries(hnu_drive_watchdog ignition-gazebo6::ignition-gazebo6 ignition-plugin1::register)
install(TARGETS hnu_drive_watchdog LIBRARY DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/gazebo_controller_adapter scripts/coordinator_gazebo_demo
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_adapter_core test/test_adapter_core.py)
  ament_add_pytest_test(test_world_builder test/test_world_builder.py)
endif()
ament_package()
