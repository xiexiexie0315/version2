cmake_minimum_required(VERSION 3.8)
project(hnu2027_sentry_autonomy)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(hnu_sentry_interfaces REQUIRED)
find_package(hnu_tunnel_core REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
ament_python_install_package(${PROJECT_NAME})
add_executable(tunnel_mission_node src/tunnel_mission_node.cpp)
target_compile_features(tunnel_mission_node PUBLIC cxx_std_17)
target_link_libraries(tunnel_mission_node hnu_tunnel_core::hnu_tunnel_core)
ament_target_dependencies(tunnel_mission_node rclcpp rclcpp_action nav2_msgs nav_msgs sensor_msgs lifecycle_msgs geometry_msgs
  std_msgs hnu_sentry_interfaces hnu_tunnel_core tf2 tf2_ros)
install(TARGETS tunnel_mission_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/simulation_sensors scripts/submit_tunnel_goal scripts/validate_autonomy DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
 find_package(ament_cmake_pytest REQUIRED)
 ament_add_pytest_test(test_autonomy_geometry test/test_autonomy_geometry.py)
 ament_add_pytest_test(test_sensor_contracts test/test_sensor_contracts.py)
endif()
ament_package()
